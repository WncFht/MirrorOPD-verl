def tmpdir():
    """Temporary directory under $TMPDIR or the session scratchpad (never /tmp)."""
    os.makedirs(SCRATCH, exist_ok=True)
    return tempfile.TemporaryDirectory(dir=SCRATCH)


def synthetic_bank(name, n_prompts, rng, empty=()):
    """Rollout rows (sample 0 plus a sample-1 decoy) with random prompt/response lengths."""
    rows = []
    for k in range(n_prompts):
        p, r = int(rng.integers(5, 30)), 0 if k in empty else int(rng.integers(3, 200))
        base = {"id": f"dapo-{k}", "prompt_ids": list(range(100 + k, 100 + k + p)), "acc": bool(rng.random() < .4),
                "finish_reason": "length" if rng.random() < .2 else "stop"}
        rows.append(base | {"sample": 0, "response_ids": rng.integers(0, 1000, r).tolist(), "n_tokens": r})
        rows.append(base | {"sample": 1, "response_ids": [1, 2, 3], "n_tokens": 3})
    return rows


def write_scores(directory, rows, value):
    """Scorer-format scores of a bank: responses.jsonl in bank order (empty responses skipped) + tokens.npz;
    the metric of response token t of prompt k is value(k, t)."""
    directory.mkdir(parents=True)
    kept = [r for r in rows if r["response_ids"]]
    with open(directory / "responses.jsonl", "w") as h:
        for r in kept:
            h.write(json.dumps({"id": r["id"], "sample": r["sample"], "n_tokens": len(r["response_ids"])}) + "\n")
    per = [np.array([value(ext.prompt_index(r["id"]) + 1000 * r["sample"], t) for t in range(len(r["response_ids"]))],
                    np.float32) for r in kept]
    offsets = np.cumsum([0] + [len(x) for x in per])
    np.savez_compressed(directory / "tokens.npz", offsets=offsets, overlap_mass=np.concatenate(per),
                        reverse_kl=-np.concatenate(per))


def build_signatures(path, banks, dim=64, seed=0):
    """Signatures in the extractor's format: prompt-specific centres, a bank offset, drift along j; the
    prompt-end state is identical across banks (same prefix)."""
    rng = np.random.default_rng(seed)
    centres, offsets = rng.normal(size=(256, dim)), {b: rng.normal(size=dim) for b in banks}
    feats, meta = [], {k: [] for k in ["bank", *ext.META_INT, "acc", "finish_reason"]}
    for b, rows in banks.items():
        for row in rows:
            if row["sample"] or not row["response_ids"]:
                continue
            plan = ext.plan_states(row)
            k = plan["prompt_index"][0]
            for a in plan["response_action"]:
                f = centres[k] + (0 if a == 0 else offsets[b] * .7 + a / 60 + rng.normal(scale=.3, size=dim))
                feats.append(f)
            meta["bank"] += [b] * len(plan["hidden_index"])
            for key, v in plan.items():
                meta[key] += v
    ext.save_signatures(path, np.asarray(feats, np.float16), meta)


class ExtractorTests(unittest.TestCase):
    def test_plan_states_positions(self):
        """Hidden index = prompt_tokens - 1 + response_action; short responses give fewer states."""
        plan = ext.plan_states({"id": "dapo-7", "prompt_ids": [1] * 5, "response_ids": [9] * 3, "acc": True,
                                "finish_reason": "stop"})
        self.assertEqual(plan["hidden_index"], [4, 5, 6])
        self.assertEqual(plan["response_action"], [0, 1, 2])
        self.assertEqual(plan["state_rank"], [0, 1, 2])
        long = ext.plan_states({"id": "dapo-0", "prompt_ids": [1] * 10, "response_ids": [2] * 700, "acc": False,
                                "finish_reason": "length"})
        self.assertEqual(len(long["hidden_index"]), 8)
        self.assertEqual(long["response_action"][0], 0)
        self.assertEqual(long["response_action"][-1], 699)
        np.testing.assert_array_equal(np.array(long["hidden_index"]) - 9, long["response_action"])

    def test_load_rows_filters(self):
        """Only sample 0 and prompt index <= max, sorted by prompt index."""
        rng = np.random.default_rng(1)
        rows = synthetic_bank("t", 10, rng)[::-1]
        with tmpdir() as d:
            p = Path(d) / "rollouts.jsonl"
            p.write_text("".join(json.dumps(r) + "\n" for r in rows))
            kept = ext.load_rows(p, 6)
        self.assertEqual([r["id"] for r in kept], [f"dapo-{k}" for k in range(7)])
        self.assertTrue(all(r["sample"] == 0 for r in kept))

    def test_refuses_overwrite(self):
        """Existing .npz or manifest: FileExistsError before anything else (no torch needed)."""
        with tmpdir() as d:
            for existing in ("sig.npz", "sig.json"):
                out = Path(d) / existing
                out.write_text("x")
                argv = ["x", "--teacher", d, "--bank", f"a={d}/r.jsonl", "--output", str(Path(d) / "sig.npz")]
                with mock.patch.object(sys, "argv", argv), self.assertRaises(FileExistsError):
                    ext.main()
                out.unlink()


class AnalysisTests(unittest.TestCase):
    def test_coverage_known_clusters(self):
        self.assertEqual(alm.coverage(np.array([0, 0, 3, 5, 5]), 10), 0.3)
        self.assertEqual(alm.coverage(np.arange(200), 200), 1.0)

    def test_coverage_draws_equal_budget(self):
        """Bank a: prompt k -> cluster k (32 of 64 prompts drawn -> exactly 0.5); bank b: one cluster."""
        prompt = np.tile(np.arange(64), 2)
        bank = np.repeat(["a", "b"], 64)
        labels = np.where(bank == "a", prompt, 0)
        res = alm.coverage_draws(labels, prompt, {"a": (bank == "a", range(64)), "b": (bank == "b", range(64))},
                                 64, n_prompts=32, n_draws=30, seed=0)
        self.assertEqual(res["a"]["mean"], 0.5)
        self.assertEqual(res["a"]["sd"], 0.0)
        self.assertAlmostEqual(res["b"]["mean"], 1 / 64)
        self.assertEqual(len(res["a"]["values"]), 30)

    def test_coverage_draws_shared_draw(self):
        """Groups with the same pool see the same prompts in each draw (paired)."""
        prompt = np.tile(np.arange(64), 2)
        bank = np.repeat(["a", "b"], 64)
        labels = prompt.copy()
        res = alm.coverage_draws(labels, prompt, {"a": (bank == "a", range(64)), "b": (bank == "b", range(64))},
                                 64, n_prompts=8, n_draws=5, seed=3)
        self.assertEqual(res["a"]["values"], res["b"]["values"])

    def test_balanced_indices(self):
        bank = np.array(["x"] * 50 + ["y"] * 20 + ["z"] * 35)
        idx = alm.balanced_indices(bank, ["x", "y", "z"], seed=42)
        self.assertEqual(len(idx), 60)
        self.assertEqual(len(set(idx.tolist())), 60)
        for b in "xyz":
            self.assertEqual((bank[idx] == b).sum(), 20)
        np.testing.assert_array_equal(np.sort(idx[bank[idx] == "y"]), np.flatnonzero(bank == "y"))
        np.testing.assert_array_equal(idx, alm.balanced_indices(bank, ["x", "y", "z"], seed=42))
        self.assertFalse(np.array_equal(idx, alm.balanced_indices(bank, ["x", "y", "z"], seed=7)))

    def test_js_divergence(self):
        p = np.array([0.2, 0.3, 0.5])
        self.assertAlmostEqual(alm.js_divergence(p, p), 0.0)
        self.assertAlmostEqual(alm.js_divergence([1, 0], [0, 1]), 1.0)
        # JS([1,0],[.5,.5]) = 0.5*KL([1,0]||[.75,.25]) + 0.5*KL([.5,.5]||[.75,.25]) in bits
        expect = .5 * np.log2(1 / .75) + .5 * (.5 * np.log2(.5 / .75) + .5 * np.log2(.5 / .25))
        self.assertAlmostEqual(alm.js_divergence([1, 0], [1, 1]), expect)
        self.assertAlmostEqual(alm.js_divergence([3, 1], [1, 2]), alm.js_divergence([1, 2], [3, 1]))

    def test_score_join_at_response_action(self):
        """The state before response token a gets token a's score; an off-by-one join fails this test."""
        rng = np.random.default_rng(2)
        rows = synthetic_bank("s", 12, rng, empty={4})
        value = lambda k, t: 10000 * k + t
        with tmpdir() as d:
            write_scores(Path(d) / "scores", rows, value)
            scores = alm.load_token_scores(Path(d) / "scores")
        self.assertNotIn(4, scores)  # empty response: no scores, later prompts still aligned
        plans = [ext.plan_states(r) for r in rows if r["sample"] == 0 and r["response_ids"]]
        meta = {k: np.array(sum((p[k] for p in plans), [])) for k in plans[0]}
        meta["bank"] = np.array(["s"] * len(meta["prompt_index"]))
        got = alm.attach_scores(meta, {"s": scores})
        expect = 10000 * meta["prompt_index"] + meta["response_action"]
        np.testing.assert_array_equal(got["overlap_mass"], expect)
        np.testing.assert_array_equal(got["reverse_kl"], -expect)
        np.testing.assert_array_equal(meta["hidden_index"] - (meta["prompt_tokens"] - 1), meta["response_action"])
        self.assertFalse(np.array_equal(got["overlap_mass"], expect + 1))

    def test_score_join_rejects_wrong_bank(self):
        """Scores whose response lengths do not match the signature's response are refused."""
        rng = np.random.default_rng(3)
        rows = synthetic_bank("s", 5, rng)
        other = [r | {"response_ids": r["response_ids"] + [0]} for r in rows]
        with tmpdir() as d:
            write_scores(Path(d) / "scores", other, lambda k, t: t)
            scores = alm.load_token_scores(Path(d) / "scores")
        plans = [ext.plan_states(r) for r in rows if r["sample"] == 0]
        meta = {k: np.array(sum((p[k] for p in plans), [])) for k in plans[0]}
        meta["bank"] = np.array(["s"] * len(meta["prompt_index"]))
        with self.assertRaises(AssertionError):
            alm.attach_scores(meta, {"s": scores})

    def test_cluster_table(self):
        """Shares per bank, dominant bank by share, separately scored bank kept out of base means."""
        labels = np.array([0, 0, 1, 1, 1, 0, 2, 2])
        bank = np.array(["a", "a", "a", "a", "b", "b", "d", "d"])
        ov = np.array([.1, .3, .5, .7, .9, .2, 5., 7.])
        t = alm.cluster_table(labels, bank, ["a", "b", "d"], 3, {"overlap_mass": ov}, ["d"])
        np.testing.assert_array_equal(t["counts"], [[2, 2, 0], [1, 1, 0], [0, 0, 2]])
        np.testing.assert_allclose(t["overlap_mass_base"][:2], [.2, .7])
        self.assertTrue(np.isnan(t["overlap_mass_base"][2]))
        self.assertAlmostEqual(t["overlap_mass_d"][2], 6.0)
        self.assertEqual(t["dominant"][2], 2)

    def test_prompt_end_check(self):
        f = np.array([[1., 0], [1., 0], [0., 1], [1., 1]])
        meta = {"response_action": np.array([0, 0, 0, 5]), "prompt_index": np.array([3, 3, 4, 3])}
        self.assertAlmostEqual(alm.prompt_end_check(f, meta, None), 0.0)
        meta["prompt_index"] = np.array([3, 3, 3, 3])
        self.assertAlmostEqual(alm.prompt_end_check(f, meta, None), 1.0)


class EndToEndTests(unittest.TestCase):
    def test_main_writes_outputs_and_refuses_overwrite(self):
        """Full analysis on synthetic signatures shaped like the real run (6 banks, 128/64 prompts)."""
        rng = np.random.default_rng(0)
        sizes = {"student": 128, "fused025": 64, "fused050": 128, "fused075": 64, "teacher": 128, "distilled": 128}
        banks = {b: synthetic_bank(b, n, rng, empty={70} if b == "student" else ()) for b, n in sizes.items()}
        with tmpdir() as d:
            d = Path(d)
            build_signatures(d / "sig.npz", banks)
            for b in ("student", "teacher", "distilled"):
                write_scores(d / f"scores-{b}", banks[b], lambda k, t: (k % 7) / 7 + t * 1e-4)
            argv = ["x", "--signatures", str(d / "sig.npz"), "--output", str(d / "out"),
                    *sum((["--scores", f"{b}={d}/scores-{b}"] for b in ("student", "teacher", "distilled")), [])]
            with mock.patch.object(sys, "argv", argv), mock.patch("builtins.print"):
                alm.main()
            res = json.loads((d / "out" / "latent_map.json").read_text())
            for fig in ("fig_small_multiples", "fig_trajectories", "fig_cluster_occupancy", "fig_coverage_js"):
                self.assertGreater((d / "out" / f"{fig}.png").stat().st_size, 10000, fig)
            self.assertTrue((d / "out" / "summary.md").read_text().startswith("# Latent state map"))
            n = sum(len(ext.plan_states(r)["hidden_index"]) for rows in banks.values() for r in rows
                    if r["sample"] == 0 and r["response_ids"])
            self.assertEqual(len(res["states"]["x"]), n)
            self.assertEqual(set(res["coverage_anchored"]), {"50", "100", "200"})
            cov = res["coverage_anchored"]["200"]
            self.assertIn("teacher (held-out)", cov)
            self.assertTrue(all(0 < v["mean"] <= 1 for v in cov.values()))
            self.assertEqual(res["banks"]["teacher"]["js_to_teacher_common"], 0.0)
            self.assertLess(res["checks"]["prompt_end_max_cosine_distance"], 1e-3)
            self.assertEqual(res["checks"]["pooled_states_per_bank"], min(
                res["banks"][b]["n_states"] for b in sizes))
            self.assertEqual(len(res["clusters"]), 200)
            self.assertIsNone(res["banks"]["fused050"]["mean_overlap_mass"])
            self.assertEqual(res["banks"]["distilled"]["scored_with"], "own (separately scored) student")
            self.assertEqual(len(res["settings"]["trajectory_prompts"]), 4)
            with mock.patch.object(sys, "argv", argv), self.assertRaises(FileExistsError):
                alm.main()
