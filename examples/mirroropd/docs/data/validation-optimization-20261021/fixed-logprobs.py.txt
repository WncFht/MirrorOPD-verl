import os
os.environ['CUDA_VISIBLE_DEVICES']='0';os.environ['OMP_NUM_THREADS']='1';os.environ['VLLM_WORKER_MULTIPROC_METHOD']='spawn';os.environ['TOKENIZERS_PARALLELISM']='false'
import json,time,statistics
from pathlib import Path
import pyarrow.parquet as pq
from transformers import AutoTokenizer
from vllm import LLM,SamplingParams
def main():
    root=Path('/mnt/tidalfs-bdsz01/dataset/llm_dataset/zkn_data/fht_data/MirrorOPD-verl');out=root/'outputs/local-validation-20261021/fixed-logprobs-v2';out.mkdir(exist_ok=False)
    model=str(root.parent/'models/DeepSeek-R1-Distill-Qwen-1.5B');tok=AutoTokenizer.from_pretrained(model,local_files_only=True)
    rows=[]
    for name in ['MATH-500','AMC23','AIME25']:rows+=pq.read_table(root/'outputs/local-validation-20261021/subsets'/(name+'.parquet')).to_pylist()
    inputs=[{'prompt_token_ids':tok.apply_chat_template(r['prompt'],tokenize=True,add_generation_prompt=True,return_dict=False)} for r in rows]
    llm=LLM(model=model,dtype='bfloat16',tensor_parallel_size=1,max_model_len=35841,max_num_seqs=32,max_num_batched_tokens=8192,gpu_memory_utilization=.5,enforce_eager=False,enable_prefix_caching=True,enable_chunked_prefill=True,logprobs_mode='processed_logprobs',compilation_config={'cudagraph_capture_sizes':[1,2,4,8,16,32]},seed=42,disable_log_stats=False)
    def generate(logprobs,tokens):
     ps=[SamplingParams(temperature=.7,top_p=.9,top_k=-1,seed=1000+i,max_tokens=tokens,ignore_eos=True,detokenize=False,logprobs=0 if logprobs else None) for i in range(32)]
     t=time.perf_counter();res=llm.generate(inputs,ps,use_tqdm=False);seconds=time.perf_counter()-t
     return seconds,[r.outputs[0].token_ids for r in res],all(r.outputs[0].logprobs is None for r in res)
    generate(True,64);generate(False,64);measurements=[];reference=None
    for i,flag in enumerate([True,False,False,True,True,False]):
     seconds,ids,absent=generate(flag,2048)
     assert len(ids)==32 and all(len(x)==2048 for x in ids)
     assert absent == (not flag)
     if reference is None:reference=ids
     else:assert ids==reference,'Fixed-seed token IDs changed'
     row={'logprobs':flag,'seconds':seconds,'output_tokens':65536,'output_tokens_per_second':65536/seconds,'token_ids_equal_to_reference':True,'probabilities_absent':absent};measurements.append(row);print(row,flush=True)
     (out/'results.json').write_text(json.dumps(measurements,indent=2)+'\n')

if __name__ == "__main__":
    main()
