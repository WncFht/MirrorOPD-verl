"""Compare normalized kappa targets on the original 993 real, EOS-coarsened states.

These are distribution-level target interventions, not new trained models.
States are equally weighted in the summaries; the systematic position sample
is not the training token-weighted risk. Teacher-preferred is a likelihood-ratio
description, not a mathematical correctness label.
"""

import hashlib
import json
import math
from pathlib import Path
from statistics import fmean as mean

import torch

ROOT = Path(__file__).resolve().parent
ETA = 0.1
STOP = [151643, 151645]


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def target_logs(lq, lt):
    first, second = math.log1p(-ETA), math.log(ETA)
    raw = {
        "0": (1 - ETA) * lq + ETA * lt,
        "0.5": 2 * torch.logaddexp(first + lq / 2, second + lt / 2),
        "1": torch.logaddexp(first + lq, second + lt),
    }
    normalizers = {k: value.logsumexp(-1) for k, value in raw.items()}
    return {k: value - normalizers[k][:, None] for k, value in raw.items()}, normalizers


def ratio_bound(threshold):
    z_min = (1 - ETA) ** 2 + ETA**2
    return z_min * (1 - ETA + ETA * threshold) / (1 - ETA + ETA * math.sqrt(threshold)) ** 2


def check_formula(lq, lt):
    q, t = lq.exp(), lt.exp()
    logs, logz = target_logs(lq, lt)
    overlap = ((lq + lt) / 2).exp()
    affinity = overlap.sum(-1)
    z = (1 - ETA) ** 2 + ETA**2 + 2 * ETA * (1 - ETA) * affinity
    reconstructed = ((1 - ETA) ** 2 * q + 2 * ETA * (1 - ETA) * overlap + ETA**2 * t) / z[:, None]
    torch.testing.assert_close(reconstructed, logs["0.5"].exp(), rtol=1e-11, atol=1e-13)
    torch.testing.assert_close(z.log(), logz["0.5"], rtol=1e-10, atol=1e-12)
    for threshold in [10, 100]:
        selected = lt - lq >= math.log(threshold)
        if selected.any():
            ratios = (logs["1"] - logs["0.5"])[selected].exp()
            assert float(ratios.min()) >= ratio_bound(threshold) - 1e-10
    return logs, affinity, z


def main():
    torch.set_num_threads(4)
    destination = ROOT / "target-overlap-diagnostic.json"
    assert not destination.exists()
    rng = torch.Generator().manual_seed(20261102)
    for scale in [0.1, 1.0, 5.0, 20.0]:
        lq = (torch.randn(256, 7, generator=rng, dtype=torch.float64) * scale).log_softmax(-1)
        lt = (torch.randn(256, 7, generator=rng, dtype=torch.float64) * scale).log_softmax(-1)
        check_formula(lq, lt)
    sampled = [json.loads(line) for line in (ROOT / "mechanism-initial-64.jsonl").open()]
    with (ROOT / "bank.jsonl").open() as stream:
        bank = {row["request_id"]: row for line in stream for row in [json.loads(line)]}
    heads = {
        role: torch.load(ROOT / f"states/{role}/head.pt", map_location="cpu", weights_only=True)
        for role in ["anchor", "teacher"]
    }
    weights = {role: value["head"].to(torch.bfloat16).float() for role, value in heads.items()}
    hidden = {role: [] for role in heads}
    inputs = []
    for row in sampled:
        original = bank[row["request_id"]]
        identity = hashlib.sha256(
            json.dumps(
                {k: original[k] for k in ["request_id", "prompt_ids", "response_ids"]}, separators=(",", ":")
            ).encode()
        ).hexdigest()
        for role in heads:
            payload = torch.load(ROOT / f"states/{role}/{row['request_id']}.pt", map_location="cpu", weights_only=True)
            assert payload["identity"]["trajectory_sha256"] == identity
            assert payload["identity"]["model"] == heads[role]["identity"]
            hidden[role].append(payload["hidden"][row["positions"]].to(torch.bfloat16).float())
        inputs.extend(
            {"request_id": row["request_id"], "position": position, "terminal": position == row["response_length"] - 1}
            for position in row["positions"]
        )
    hidden = {role: torch.cat(values) for role, values in hidden.items()}
    keep = torch.ones(len(weights["anchor"]), dtype=torch.bool)
    keep[STOP] = False
    result = []
    prior = json.loads((ROOT / "eos-action-space-diagnostic.json").read_text())
    expected = {(r["request_id"], r["position"]): r for r in prior["states"]}
    maximum = 0.0
    with torch.inference_mode():
        for start in range(0, len(inputs), 64):
            stop = min(start + 64, len(inputs))
            lp = {}
            for role in heads:
                raw = (hidden[role][start:stop] @ weights[role].T).double().log_softmax(-1)
                lp[role] = torch.cat([raw[:, keep], raw[:, STOP].logsumexp(-1, keepdim=True)], -1)
            lq, lt = lp["anchor"], lp["teacher"]
            q, t = lq.exp(), lt.exp()
            logs, affinity, z = check_formula(lq, lt)
            targets = {k: value.exp() for k, value in logs.items()}
            baseline = (q * (lq - lt)).sum(-1)
            target_teacher_kl = {k: (targets[k] * (value - lt)).sum(-1) for k, value in logs.items()}
            rows = []
            for i in range(stop - start):
                record = inputs[start + i] | {
                    "sqrt_affinity": float(affinity[i]),
                    "half_normalizer": float(z[i]),
                    "half_components": {
                        "q": float((1 - ETA) ** 2 / z[i]),
                        "normalized_overlap": float(2 * ETA * (1 - ETA) * affinity[i] / z[i]),
                        "teacher": float(ETA**2 / z[i]),
                    },
                    "q_stop": float(q[i, -1]),
                    "teacher_stop": float(t[i, -1]),
                    "target_stop": {k: float(value[i, -1]) for k, value in targets.items()},
                    "q_teacher_kl": float(baseline[i]),
                    "target_teacher_kl": {k: float(value[i]) for k, value in target_teacher_kl.items()},
                    "teacher_preferred": {},
                }
                maximum = max(
                    maximum,
                    abs(
                        record["q_teacher_kl"]
                        - expected[(record["request_id"], record["position"])]["coarse_teacher_kl"]
                    ),
                )
                rows.append(record)
            for threshold in [10, 100]:
                mask = lt - lq >= math.log(threshold)
                quantities = {"q_mass": (q * mask).sum(-1), "teacher_mass": (t * mask).sum(-1)}
                quantities |= {f"target_{k}_mass": (value * mask).sum(-1) for k, value in targets.items()}
                for i, record in enumerate(rows):
                    record["teacher_preferred"][str(threshold)] = {
                        key: float(value[i]) for key, value in quantities.items()
                    }
            result.extend(rows)
            print("Target overlap", stop, "/", len(inputs), flush=True)
    assert len(result) == 993 and maximum < 1e-6
    summaries = {}
    groups = {
        "all_sampled_states": result,
        "terminal_states": [r for r in result if r["terminal"]],
        "teacher_stop_ratio_ge10_mass_ge001": [
            r for r in result if r["teacher_stop"] >= 0.01 and r["teacher_stop"] >= 10 * r["q_stop"]
        ],
    }
    for name, rows in groups.items():
        if not rows:
            summaries[name] = {"states": 0}
            continue
        summaries[name] = {
            "states": len(rows),
            "sqrt_affinity": mean([r["sqrt_affinity"] for r in rows]),
            "half_components": {
                key: mean([r["half_components"][key] for r in rows]) for key in ["q", "normalized_overlap", "teacher"]
            },
            "q_stop": mean([r["q_stop"] for r in rows]),
            "teacher_stop": mean([r["teacher_stop"] for r in rows]),
            "target_stop": {k: mean([r["target_stop"][k] for r in rows]) for k in ["0", "0.5", "1"]},
            "teacher_preferred": {
                str(threshold): {
                    key: mean([r["teacher_preferred"][str(threshold)][key] for r in rows])
                    for key in rows[0]["teacher_preferred"][str(threshold)]
                }
                for threshold in [10, 100]
            },
        }
    report = {
        "scope": __doc__,
        "eta": ETA,
        "states": result,
        "summaries": summaries,
        "ratio_bounds": {str(t): ratio_bound(t) for t in [10, 100]},
        "formula_checks": "1024 random seven-action distributions plus all993 real coarsened states",
        "prior_cpu_coarse_kl_max_difference": maximum,
        "script_sha256": digest(Path(__file__)),
        "source_sha256": {
            name: digest(ROOT / name)
            for name in ["mechanism-initial-64.jsonl", "bank.jsonl", "eos-action-space-diagnostic.json"]
        },
        "limits": [
            "One initial model and one sampled bank; no optimizer or model update is performed.",
            "The overlap component also contains teacher information; its teacher coefficient is not supervision share.",
            "Teacher-preferred likelihood ratios do not label actions as mathematically correct.",
            "Equal sampled-state means are not full training-token averages.",
            "EOS strata are exploratory and correlated within trajectories; no population significance claim.",
        ],
    }
    destination.write_text(json.dumps(report, indent=2) + "\n")
    print(json.dumps(summaries, indent=2))


if __name__ == "__main__":
    main()
